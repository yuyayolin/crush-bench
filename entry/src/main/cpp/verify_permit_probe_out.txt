  [旧] 权威 = sched_getaffinity(0-8) / SMT 关
       生效序列 = 7 核 cpu4,cpu6,cpu8,cpu0,cpu1,cpu2,cpu3
       快簇 = 3 核 cpu4,cpu6,cpu8(@2270MHz)   <- 真机 14:24 之后那一行'3 核簇/最高2270MHz'
  [新] 权威 = 实测探测(0-13) / SMT 开  -> authority=0
       快簇 = 2 核 cpu12,cpu13(@2750MHz)   <- 就是全机最快档
  [新] 权威 = 实测探测(0-13) / SMT 关
       快簇 = 4 核 cpu12,cpu4,cpu6,cpu8(最高 2750MHz)

  逐核探测判定(rc = sched_setaffinity 返回值; back = 立刻读回的掩码):
    内核接受该核                 -> accept (errno=0)
    EINVAL(不在 cpuset 内)    -> reject (errno=22)
    EPERM(无权改亲和性)          -> reject (errno=1)
    返回 0 但读回被夹回            -> reject (errno=-4)
    返回 0 但读回失败             -> reject (errno=-1)
